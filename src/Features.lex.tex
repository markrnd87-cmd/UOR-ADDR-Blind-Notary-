def isFeatureWellBacked (s : FeatureSpec) : Bool :=
  (!s.stakeholders.isEmpty) &&
  (!s.concerns.isEmpty) &&
  (!s.registerIds.isEmpty) &&
  (!s.backingTheorems.isEmpty) &&
  (!s.freeCommonsBinary.isEmpty)

theorem allFeaturesBackedByKernelClaims (f : FeatureId) :
    isFeatureWellBacked (specFor f) = true := by
  cases f <;> rfl

theorem everyConcernAddressedByFeature (c : Concern) :
    allFeatures.any (fun f => ((specFor f).concerns).contains c) = true := by
  cases c <;> rfl
